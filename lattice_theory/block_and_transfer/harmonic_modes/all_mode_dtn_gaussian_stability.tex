\[
 |Q_B(v,v)|\le a_R\,\langle v,S_Bv\rangle,
 \qquad a_R=C\log(2+R).                                      \tag{1}
\]
Here $S_B$ is the ORIGINAL classical Dirichlet-to-Neumann (Schur)

 \|S_B^{-1/2}Q_BS_B^{-1/2}\|\le a_R,
 \quad |Q_B(u,v)|\le a_R\|u\|_{S_B}\|v\|_{S_B}.               \tag{2}
\]

positivity of the signed loop coefficient.

\paragraph{1. A joint form with controlled overlap.}
Use a genuine common relative Gaussian/forest and disjoint block
interiors $I$, with retained skeleton $Y$. The original precision $A$
has exact Schur form
\[
 S=A_{YY}-A_{YI}A_{II}^{-1}A_{IY},\qquad

\]
For a complete field $w$, let $v_B(w)$ be its literal block boundary
restriction. Define only the DECLARED local kinetic piece
\[
 Q(w,z)=\sum_B Q_B(v_B(w),v_B(z)).                             \tag{4}
\]
Its harmonic energy is no larger than the complete block energy of
that restriction. If complete plaquette sets overlap at most $q$, then
\[
 |Q(w,w)|\le a_R\sum_B\mathcal E_B(w)
                  \le \alpha\mathcal E(w),\quad \alpha=qa_R. \tag{5}
\]

This bound holds on the full energy quotient, hence also bilinearly.
It involves no number of blocks. Distinct blocks are not independent.

Let $Ey$ be the original harmonic lift of skeleton data. Then
$\mathcal E(Ey)=y^*Sy$, and $Q(Ey,Ey)=y^*My$ for a symmetric $M$.

\[
 -\alpha S\le M\le\alpha S,

\]
